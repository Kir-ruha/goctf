cmake_minimum_required(VERSION 3.20)
project(lighthouse LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(lighthouse main.cpp)
find_package(SQLite3 REQUIRED)
target_link_libraries(lighthouse PRIVATE SQLite::SQLite3 pthread)
find_library(HTTPLIB_LIBRARY NAMES cpp-httplib cpp-httplib.so.0.11 cpp-httplib.so.0.14 REQUIRED)
target_link_libraries(lighthouse PRIVATE ${HTTPLIB_LIBRARY})
find_library(WASMTIME_LIBRARY NAMES wasmtime REQUIRED)
target_link_libraries(lighthouse PRIVATE ${WASMTIME_LIBRARY})
